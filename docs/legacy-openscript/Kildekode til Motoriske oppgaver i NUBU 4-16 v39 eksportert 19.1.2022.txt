to get yearDecToMonth REAL theIn
	local STRING output
	local INT vYear
	local INT vMonth
	vYear = truncate(theIn)
	vMonth = truncate(((theIn-vYear))*12)
	return (vYear & ":" & vMonth)
end yearDecToMonth

to get yearMonthToDec STRING theIn
	local LONG ctrJ
	step ctrJ from 1 to charCount(theIn)
		if char ctrJ of theIn is not in "01234567890:" then
			set sysError to ("Invalid input for yearMonthToDec: " & theIn)
			return NULL
		end if
	end step
	local LONG colonOffset
	get offset (":", theIn)
	if It is 0 then
		set sysError to "Missing : in input for yearMonthToDec"
		return NULL
	else
		colonOffset = It
	end if
	-- förutsätter att input är OK nu.
	local INT vYear
	local INT vMonth
	vYear = (chars 1 to (colonOffset-1) of theIn)
	vMonth = (chars (colonOffset+1) to charCount(theIn) of theIn)
	local REAL output
	output = vYear + (vMonth/12) + 1/24   -- därför att i genomsnitt 1/2 månad försvinner vid trunkeringen
	return output
end yearMonthToDec

to get estimateToPresentText REAL theIn
	local REAL ageEquivalent
	ageEquivalent = exp((text of field "param1" of page "Calculate")+((\
	text of field "param2" of page "Calculate")*theIn))
	local STRING output
	conditions
		when ageEquivalent < (text of field "minest" of page "Calculate")
			output = "Lavere enn " & yearDecToMonth(text of field "minest" of page "Calculate")
		when ageEquivalent > (text of field "maxest" of page "Calculate")
			output = "Høyere enn " & yearDecToMonth(text of field "maxest" of page "Calculate")
		else -- value within range
			output = yearDecToMonth(ageEquivalent)
	end conditions
	return output
end estimateToPresentText

to get validDateSecondsFromNorwegian STRING theIn
	get sysDate
	format date it as "yyyy-mm-dd" from sysDateFormat
	local STRING thisYear
	thisYear = chars 1 to 4 of It
	get theIn
	if isType(date,It,"d.m.") is TRUE then
		get (It & thisYear)
		format date It as seconds from "d.m.yyyy"
		return It
	else
		if isType(date,It,"dd.mm.") is TRUE then
			get (It & thisYear)
			format date It as seconds from "dd.mm.yyyy"
			return It
		else
			if isType(date,It,"d.m.yy") is TRUE then
				format date It as seconds from "d.m.yy"
				return It
			else
				if isType(date,It,"dd.mm.yy") is TRUE then
					format date It as seconds from "dd.mm.yy"
					return It
				else
					if isType(date,It,"d.m.yyyy") is TRUE then
						format date It as seconds from "d.m.yyyy"
						return It
					else
						if isType(date,It,"dd.mm.yyyy") is TRUE then
							format date It as seconds from "dd.mm.yyyy"
							return It
						else
							return -1
						end if
					end if
				end if
			end if
		end if
	end if
end

to get modellpersentilFraEstimat REAL theIn
	--theIn skal nå rundes av til nærmeste multipel av 0.05 for å 
	--tilsvare tabellen.
	theIn = round(theIn*20)/20
	local LONG ctrJ
	local aModellpersentiler[][]
	local WORD antallTabellrader
	aModellpersentiler = pModellpersentiler of self
	antallTabellrader = (item 1 of dimensions(aModellpersentiler))
	if theIn <= aModellpersentiler[1][1] then
		return aModellpersentiler[1][2]
	end if
	ctrJ = 2
	while theIn > aModellpersentiler[ctrJ][1] and (ctrJ < antallTabellrader)
		increment ctrJ
	end while
	return aModellpersentiler[ctrJ][2] -- den siste må være 100.
end

to get aldersgruppepersentilerFraEstimat REAL theIn
	local aAldersgruppepersentiler[][]
	aAldersgruppepersentiler = pAldersgruppepersentiler of self
	local STACK whatGroup -- can be 1 to 4 depending on limits
	local LONG ctrJ
	step ctrJ from (item 1 of dimensions(aAldersgruppepersentiler)) to 1 by -1
		if theIn <= aAldersgruppepersentiler[ctrJ][2] then
			push 1 onto WhatGroup
			continue step ctrJ
		else
			if theIn <= aAldersgruppepersentiler[ctrJ][3] then
				push 2 onto WhatGroup
				continue step ctrJ
			else
				if theIn <= aAldersgruppepersentiler[ctrJ][4] then
					push 3 onto WhatGroup
					continue step ctrJ
				else
					push 4 onto WhatGroup --i.e., over 85th percentile.
				end if
			end if
		end if
	end step
	--konstruerar textoutput
	local STRING output
	textline ((item 1 of dimensions(aAldersgruppepersentiler))+2) of output = " "
	local WORD strlen1,strlen2,strlen3,strlen4,strlen5
	strlen1 = 17
	strlen2 = 12
	strlen3 = 12
	strlen4 = 12
	strlen5 = 12
	put LText("Aldersgruppe",strlen1) & \
	CText("-15.",strlen2) & \
	CText(">15.-50.",strlen3) & \
	CText(">50.-85.",strlen4) & \
	CText(">85.",strlen5) \
	into textline 1 of output
	put LText("========================",strlen1) &&  \
	RText("========================",strlen2-1) && \
	RText("========================",strlen3-1) && \
	RText("========================",strlen4-1) && \
	RText("========================",strlen5-1) \
	into textline 2 of output
	step ctrJ from 1 to (item 1 of dimensions(aAldersgruppepersentiler))
		put LText(aAldersgruppepersentiler[ctrJ][1],strlen1) into textline (ctrJ+2) of output
		if item ctrJ of WhatGroup = 1 then
			put CText("X",strlen2) after textline (ctrJ+2) of output
		else
			put LText(NULL,strlen2) after textline (ctrJ+2) of output
		end if
		if item ctrJ of WhatGroup = 2 then
			put CText("X",strlen3) after textline (ctrJ+2) of output
		else
			put LText(NULL,strlen3) after textline (ctrJ+2) of output
		end if
		if item ctrJ of WhatGroup = 3 then
			put CText("X",strlen4) after textline (ctrJ+2) of output
		else
			put LText(NULL,strlen4) after textline (ctrJ+2) of output
		end if
		if item ctrJ of WhatGroup = 4 then
			put CText("X",strlen5) after textline (ctrJ+2) of output
		else
			put LText(NULL,strlen5) after textline (ctrJ+2) of output
		end if
	end step
	put (CRLF & LText(NULL,strlen1) & \
	CText("- - - - - - Faktisk persentil i referansegruppen - - - - - -",(strlen2+strlen3+strlen4+strlen5)) \
	& CRLF) before output
	put CRLF after output
	return output
end

to get LText STRING theIn, WORD nofChars
	return chars 1 to nofChars of (theIn & "                                                                                                    ")
	-- 100 chars
end

to get RText STRING theIn, WORD nofChars
	get ("                                                                                                    " & theIn) 
	-- 100 chars
	local charCountString
	charCountString=charCount(It)
	return (chars (charCountString-nofChars+1) to charCountString of It)
end

to get CText STRING theIn, WORD nofChars
	get ("                                                  " & theIn & "                                                  ") 
	-- 50 chars + theIn + 50 chars
	local charCountString
	charCountString=charCount(It)
	return (chars 1 to nofChars of \
	(chars (round((charCountString-nofChars)/2)+1) to charCountString of It))
end

to get writeNUBUFile STRING inFileName
	--filformat: Rader NYTT FRAAN JAN06: Rad9 är en versionssträng.
	--1. Navn
	--2. checked of button "gutt" && checked of button "jente"
	--3. Fodtdato
	--4. Testdato
	--5. Alder
	--6. Merknad
	--7. Merknad2
	--8. 50 items (tas fra punchingside)
	--9. Versionssträng "NUBUMotor_100" för denna version (1.0.0, januari 2006)
	sysSuspend = FALSE
	closeFile inFileName
	sysError = NULL
	createFile inFileName
	if sysError <> NULL and sysError <> "ok" then
		return 0
	end if
	sysError = NULL
	local LONG ctrJ
	local STRING dataString
	step ctrJ from 1 to 50
		get ("i" & ctrJ)
		if text of field (It) of page "Start" of self = NULL then
			put "*" after \
			dataString
		else
			put char 1 of ((text of field (It) of page "Start" of self) & "9") after \
			dataString
		end if
	end step
	get ((text of field "Navn" of page "Start" of self) & CRLF & \
	(checked of button "gutt" of page "Start" of self) && \
	(checked of button "jente" of page "Start" of self) & CRLF & \
	(text of field "Fodtdato" of page "Start" of self) & CRLF & \
	(text of field "Testdato" of page "Start" of self) & CRLF & \
	(text of field "Alder" of page "Start" of self) & CRLF & \
	(text of field "Merknad" of page "Start" of self) & CRLF & \
	(text of field "Merknad2" of page "Start" of self) & CRLF & \
	(datastring) & CRLF & \
	"NUBUMotor_100" & CRLF)
	sysError = NULL
	writeFile It to inFileName
	if sysError <> NULL and sysError <> "ok" then
		return 0
	end if
	sysError = NULL
	closeFile inFileName
	if sysError <> NULL and sysError <> "ok" then
		return 0
	end if
	sysSuspend = TRUE
	return 1 -- success
end

to get readNUBUFile STRING inFileName
	--filformat: Rader
	--1. Navn
	--2. checked of button "gutt" && checked of button "jente"
	--3. Fodtdato
	--4. Testdato
	--5. Alder
	--6. Merknad
	--7. Merknad2
	--8. 50 items (på punchingside)
	--9. Versionssträng "NUBUMotor_100" för denna version (1.0.0, januari 2006)
	sysSuspend = FALSE
	closeFile inFileName
	sysError = NULL
	openFile inFileName
	if sysError <> NULL and sysError <> "ok" then
		return 0
	end if
	readFile inFileName to eof -- kan inte vara > 32000 ngt.
	if sysError <> NULL and sysError <> "ok" then
		return 0
	end if
	sysError = NULL
	if (textline 9 of It) <> "NUBUMotor_100" then
		request "Ukjent filformat. Vil du fortsette?" with "&Fortsett" or "&Avbryt"
		if It is not "Fortsett" then
			closeFile inFileName
			break to system
		else
			--må lese filen igjen, for "Fortsett" har nå fylt opp It.
			sysSuspend = FALSE
			closeFile inFileName
			sysError = NULL
			openFile inFileName
			if sysError <> NULL and sysError <> "ok" then
				return 0
			end if
			readFile inFileName to eof -- kan inte vara > 32000 ngt.
			if sysError <> NULL and sysError <> "ok" then
				return 0
			end if
			sysError = NULL
		end if
	end if
	(text of field "Navn" of page "Start" of self) = textline 1 of It
	(checked of button "gutt" of page "Start" of self) = word 1 of textline 2 of It
	(checked of button "jente" of page "Start" of self) = word 2 of textline 2 of It
	(text of field "Fodtdato" of page "Start" of self) = textline 3 of It
	(text of field "Testdato" of page "Start" of self) = textline 4 of It
	(text of field "Alder" of page "Start" of self) = textline 5 of It
	(text of field "Merknad" of page "Start" of self) = textline 6 of It
	(text of field "Merknad2" of page "Start" of self) = textline 7 of It
	local LONG ctrJ
	local STRING dataString
	dataString = textline 8 of It
	local STRING dum1
	step ctrJ from 1 to 50
		get ("i" & ctrJ)
		dum1 = char ctrJ of (datastring & \
		"99999999999999999999999999999999999999999999999999")
		if dum1 is not "*" then
			(text of field (It) of page "Start" of self) = dum1
		else
			(text of field (It) of page "Start" of self) = NULL
		end if
	end step
	send leaveField to field "fodtdato" of page "Start"
	send leaveField to field "testdato" of page "Start"
	send leaveField to field "alder" of page "Start"
	send suggestAge to page "Start"
	sysError = NULL
	closeFile inFileName
	sysError = NULL
	sysSuspend = TRUE
	return 1 -- success
end

					(text of field (dumFeltnavn) of page "Start" of self) = dum1
				else
					(text of field (dumFeltnavn) of page "Start" of self) = NULL

			if (hj_skrivSeparateFiler) then
				--skaper på navn
				dumFeltnavn = text of field "Navn" of page "Start" of self
				dum1 = NULL
				
				while (fileExists32(hj_thisDir & dumFeltnavn & dum1 & ".~Z4") = 1) or dumFeltnavn = NULL -- exists
					put char (random(22)) of "ABCDEFGHKLMNPQRSTUVXYZ___" after dum1
				end while
				currentFile = (hj_thisDir & dumFeltnavn & dum1 & ".~Z4")
				caption of mainWindow = RText(("Skåring av motoriske oppgaver i NUBU 4-16 (" & hj_shortFileName(currentFile) & ")"),77)
				currentFileIsSaved = TRUE -- will not ask
				if writeNUBUFile(currentFile) = 0 then
					request "Error3"
					break to system
				end if
			end if
			if (hj_skrivUt) then
				in mainWindow
					send openPrintPreview
				end in
				in viewer "printPreview"
					send printResults
				end in
			end if
		until FALSE
		closeFile inFileName
		sysError = NULL
		sysSuspend = TRUE
	end if	
end

to handle laesMangeFiler
	--getFileOnlyList32(<path>,<file attributes>,<sort order>)
	--chooseDirectoryDlg32(<caption text>,<default path>)
	--saveAsDlg32(<caption text>,<default file>,<default path>,<filters>,<index>)
	local STRING vDir,vFilliste,vFilters,outFile,outText,separateChar
-- Det går å la brukeren velge tabseparert el linjeseparert, men hva er poenget med linjeseparert?
--	request "Linjeseparert eller tabseparert output?" with "&Linje" or "&Tab"
--	if It is not "Tab" then
--		put CRLF into separateChar
--	else
		put tab into separateChar
--	end if
	system LOGICAL currentFileIsSaved
	vDir = chooseDirectoryDlg32("Velg katalog å lese filer fra:","")
	if vDir is NULL then
		break laesMangeFiler
	end if
	vFilliste = getFileOnlyList32((vDir & "\*.*"),"","N")
	vFilters= "Tekstfil (*.txt),*.txt,Datafil (*.dat),*.dat"
	outFile = saveAsDlg32("Lagre din datafil som:","Data.txt","",vFilters,1)
	if outFile is NULL then
		break laesMangeFiler
	end if
	-- öppnar utfil.
	closeFile outFile
	get removeFile32(outFile)	
	get yieldApp()
	get yieldApp()
	mmYield
	createFile outFile	
	local LONG ctrJ
	local writeBuffer
	
	-- skriver hoved til tabfil.
	get \
	("FILNAVN" & tab & \
	"NAVN" & tab & \
	"KJØNN" & tab & \
	"LOGALDER" & tab & \
	"MERKNADER" & tab & \
	"OPPGAVESKÅRER" & tab & \
	"RASCHSKÅRE" & tab & \
	"STANDARDFEIL" & tab & \
	"MINESTIMERT" & tab & \
	"MAKSESTIMERT" & tab & \
	"ALDERSEKVIVALENT" & tab & \
	"NEDRE GRENSE KONF.INT." & tab & \
	"ØVRE GRENSE KONF.INT." & tab & \
	"MODELLBASERT PERSENTIL" & CRLF) 
	writeFile It to outFile
	
	step ctrJ from 1 to (textlineCount(vFilliste))
		clear outText
		--Öppnar filen.
		send aapneFil "X", (vDir & "\" & (textline ctrJ of vFilliste))
		--filformat: Rader NYTT FRAAN JAN06: Rad9 är en versionssträng.
		-- Kalkylerar.
		send buttonclick "X" to button "Beregn" of page "Start" of self
		currentFileIsSaved = TRUE
		-- Skriver till datafilen följande:
		--Rad 1, filnamn (inte path)
		put (textline ctrJ of vFilliste & separateChar) after outText
		--Rad 2. Navn
		put (text of field "navn" of page "Start" & separateChar) after outText
		--Rad 3. Kjönn 0=J 1=G 9=ikke [checked of button "gutt" && checked of button "jente"]
		if checked of button "gutt" of page "Start" is FALSE then
			if checked of button "jente" of page "Start" is FALSE then
				put "9" & separateChar after outText
			else
				put "0" & separateChar after outText
			end if
		else
			put "1" & separateChar after outText
		end if
		--Rad 4. logalder
		get text of field "alder" of page "Start"
		if It is NULL then
			put "0" & separateChar after outText
		else
			writeBuffer = ln(yearMonthtoDec(It))
			-- writeBuffer = FNum(writeBuffer,8,4)
			format number writeBuffer as ("0" & sysDecimal & "000")
			put writeBuffer & separateChar after outText
		end if
		--Rad 5. Merknad && Merknad2.
		put (text of field "merknad" of page "Start" && text of field "merknad2" of page "Start") & separateChar after outText 
		--Rad 6. 50 items (fra felt på beregningsside)
		put quote & (text of field "INP" of page "Calculate" ) & quote & separateChar after outText 
		--Rad 7. Rasch-skåre
		writeBuffer = pMeasure of page "Calculate"
--		writeBuffer = FNum(writeBuffer,8,4)
		format number writeBuffer as ("0" & sysDecimal & "000")
		put writeBuffer & separateChar after outText
		--Rad 8. SE
		writeBuffer = pSE of page "Calculate"
--		writeBuffer = FNum(writeBuffer,8,4)
		format number writeBuffer as ("0" & sysDecimal & "000")
		put writeBuffer & separateChar after outText
		--Rad 9, IsMinEstimated
		if pIsMinEstimated of page "Calculate" is TRUE then
			put "1" & separateChar after outText
		else
			put "0" & separateChar after outText
		end if
		--Rad 10, IsMaxEstimated
		if pIsMaxEstimated of page "Calculate" is TRUE then
			put "1" & separateChar after outText
		else
			put "0" & separateChar after outText
		end if
		--Rad 11, aldersekvivalent i år:måneder
		put estimateToPresentText(pMeasure of page "Calculate") & separateChar after outText
		--Rad 12 & 13 nedre og øvre grense for konfidensintervall i aldersekvivalent, blank for missing
		if isType("REAL",pSE of page "Calculate") is TRUE and \
		pIsMinEstimated of page "Calculate" is FALSE and \
		pIsMaxEstimated of page "Calculate" is FALSE then
			put estimateToPresentText((pMeasure of page "Calculate") - 1.96*(pSE of page "Calculate"))  & separateChar after outText
			put estimateToPresentText((pMeasure of page "Calculate") + 1.96*(pSE of page "Calculate"))  & separateChar after outText
		else
			put separateChar after outText
			put separateChar after outText
		end if
		--14. Modellbasert persentil.
		if isType("REAL",pAlder of field "alder" of page "Start") then
			put modellpersentilFraEstimat(pMeasure of page "Calculate"- \
			ageToMeasure(pAlder of field "alder" of page "Start", \
			(text of field "param1" of page "Calculate"), \
			(text of field "param2" of page "Calculate"))) after outText
		end if
		--Ny linje.
		put CRLF after outText
		--Skriver till fil
		writeFile outText to outFile
	end step
	closeFile outFile
end laesMangeFiler

to get raschMeasureFromItems \
STRING responseString, \
STRING keyString, \
WORD nofItems, \
WORD nofCats, \
STACK aItemDifficulties, \
STACK aCatDifficulties, \
DWORD maxIterations, \
REAL convergence


	-- ==============================================================================
	-- If successful will put measure and SE, number of valid items, whether
	-- maximum or minimum estimated,
	-- as page properties in addition to in return text.
	-- pMeasure of self (i.e.,page)
	-- pSE of self
	-- pNValid of self
	-- pIsMinEstimated
	-- pIsMaxEstimated
	-- ==============================================================================
	-- Return codes other than an expected ability estimate.
	-- -10000 dummy return statement, has reached end of program without 
	--        expected result
	-- -10001 no valid completed items by this person
	-- -10002 reached maxIterations without convergence

	-- ==============================================================================
	-- This program has several limitations, which include:

	-- All categories must be 1 character wide
	-- Response categories are assigned values of 0, 1 ... (nofCats-1) in order of
	--   presentation in keyString
	-- Assumes all item and cat difficulties are given in logits
	
	-- ==============================================================================

	-- Input handling

	-- nulls page properties for results
	clear pMeasure of self
	clear pSE of self
	clear pNValid of self
	clear pIsMinEstimated of self
	clear pIsMaxEstimated of self

	-- Converts item and category difficulty calibrations to arrays
	local REAL itemDifficulties[]
	local LOGICAL thisItemCompleted[]
	local LONG ctrJ
	itemDifficulties[nofItems]=0 -- gets dimension of array right
	thisItemCompleted[nofItems]=FALSE -- gets dimension of array right
	step ctrJ from 1 to nofItems
		itemDifficulties[ctrJ] = item ctrJ of aItemDifficulties
	end step
	local REAL catDifficulties[]
	catDifficulties[nofCats]=0 -- gets dimension of array right
	step ctrJ from 1 to nofCats
		catDifficulties[ctrJ] = item ctrJ of aCatDifficulties
	end step
	
	-- Sets defaults if not given
	-- maxIterations set to 500 if 0
	if maxIterations is 0 then
		maxIterations = 500
	end if
	
	-- keyString can be empty, 0, 1 ... (nofCats-1) then assumed
	if keyString is "" then
		nofCats = chars 1 to (nofCats) of "0123456789abcdefghijklmnopqrstuvwxyz"
	end if

	-- ==============================================================================

	-- Step 1. Collect responses by person n to the desired
	-- subset of L calibrated polytomous or dichotomous items.
	-- Person n has a raw score of R.
	-- RMin is the minimum possible score on these items,
	-- RMax the maximum possible.

	-- Step 3.
	-- The average item difficulty of person n's L items is
	-- Dmean = (1/L)* Sum[from i=1 to L]of Di

	local WORD nofCompletedItems
	local REAL rawScore
	local REAL rMin
	local REAL rMax


	-- figures out the number of completed items
	local INT thisRawScore

	-- and at the same time calculated mean difficulty for completed items
	local REAL dMean
	dMean = 0

	step ctrJ from 1 to nofItems
		get getNumRawScore((char ctrJ of responseString),keyString,nofCats)
		if It > -1 then
			thisItemCompleted[ctrJ] = TRUE
			increment nofCompletedItems
			increment rawScore by It
			dMean = dMean + itemDifficulties[ctrJ]
		else
			thisItemCompleted[ctrJ] = FALSE -- should really not be needed but anyway.
		end if
	end step
	pNValid of self = nofCompletedItems
	if nofCompletedItems = 0 then
		return -10001 -- this should be the error code for no valid completed items
	end if
	rMin = 0
	rMax = nofCompletedItems*(nofCats-1)
	dMean = dMean/nofCompletedItems
	pIsMinEstimated of self = FALSE
	pIsMaxEstimated of self = FALSE

	if rawScore = rMin then
		pIsMinEstimated of self = TRUE
		rawScore = rMin + 0.3
	end if
	if rawScore = rMax then
		pIsMaxEstimated of self = TRUE
		rawScore = rMax - 0.3
	end if

	-- ==============================================================================

	-- Step 2. Each item i has a calibration Di and each step j a calibration Fj, in
	-- user-scaled units. If not already in logits, convert these to logits
	-- (Here it is assumed that all calibrations are in logits.)
	
	-- Step 3 is performed above.
	
	-- ==============================================================================

	-- Step 4. The initial estimate of person n's ability can be any finite value.
	-- Convenient ones are the mean item difficulty, a previous ability estimate, 
	-- or M=dMean+ln((rawScore-rMin)/(rMax-rawScore))
	
	local REAL oldM
	
	oldM = dMean+ln((rawScore-rMin)/(rMax-rawScore))

	put CRLF & CRLF after text of field "output" of self
	put "rawScore      = " & rawScore & CRLF after text of field "output" of self
	put "dMean         = " & dMean & CRLF after text of field "output" of self


	-- ==============================================================================

	-- Step 5. Compute expected score and variance for M. The categories for item i
	-- of difficulty Di are numbered b, ..., t. Fb=0, and the other Fk are the
	-- Rasch-Andrich step thresholds. The denominator is the sum o fhe numerators
	-- for all categories, so that the sum of the probabilities across all categories
	-- is 1:
	
	local REAL Pnij,enumPnij,denomPnij,partDenomPnij
	local REAL expectedScore,expectedScoreVariance,expectedScoreVariance1,expectedScoreVariance2
	local LONG ctrI,ctrH,ctrK,ctrIteration
	ctrIteration = 0
	do
		increment ctrIteration
		expectedScore = 0
		expectedScoreVariance = 0
		
		--För koll.
		put CRLF & "---- ITERATION " & ctrIteration & " ----" & CRLF after text of field "output" of self
		text of field "underway" of self = ("ITERATION " & ctrIteration)
		
		
		step ctrI from 1 to nofItems
			--skips if item not completed
			if thisItemCompleted[ctrI] = FALSE then
				continue step ctrI
			end if
			denomPnij = 0
			step ctrH from 1 to nofCats
				partDenomPnij = 0
				step ctrK from 1 to ctrH
					partDenomPnij=partDenomPnij+catDifficulties[ctrK]
				end step
				denomPnij=denomPnij+(exp((ctrH-1)*(oldM-itemDifficulties[ctrI])-partDenomPnij))
			end step
	
--			-- För koll.
--			put CRLF & "Var " & ctrI & ": denomPnij= " & denomPnij & CRLF after text of field "output" of self
			expectedScoreVariance1 = 0
			expectedScoreVariance2 = 0
			step ctrJ from 1 to nofCats
				enumPnij = 0
				step ctrK from 1 to ctrJ
					enumPnij=enumPnij+catDifficulties[ctrK]
				end step
				enumPnij=exp((ctrJ-1)*(oldM-itemDifficulties[ctrI])-enumPnij)
				Pnij=enumPnij/denomPnij
	
--				-- För koll.
--				put CRLF & "Cat " & ctrJ & ":" & CRLF after text of field "output" of self
--				put "enumPnij = " & enumPnij & CRLF after text of field "output" of self
--				put "Pnij     = " & Pnij & CRLF after text of field "output" of self

				expectedScore = expectedScore + ((ctrJ-1)*Pnij)
				expectedScoreVariance1=expectedScoreVariance1+\
				((ctrJ-1)^2*Pnij)
				expectedScoreVariance2=expectedScoreVariance2+\
				((ctrJ-1)*Pnij)
			end step
			expectedScoreVariance=expectedScoreVariance+\
			(expectedScoreVariance1-expectedScoreVariance2^2)
		end step
		
		-- för koll.
		put CRLF & CRLF after text of field "output" of self
		put "oldM          = " & oldM & CRLF after text of field "output" of self
		put "expectedScore = " & expectedScore & CRLF after text of field "output" of self
		put "variance      = " & expectedScoreVariance & CRLF after text of field "output" of self
		
	
		step ctrI from 1 to nofItems
			--skips if item not completed
			if thisItemCompleted[ctrI] = FALSE then
				continue step ctrI
			end if
			step ctrJ from 1 to nofCats
			end step
		end step		
	
		-- ==============================================================================
	
		-- Step 6. Obtain a better estimate M' of the measure M:
		
		newM = oldM+ (rawScore - expectedScore)/expectedScoreVariance
		if abs(newM-oldM)<convergence then
			break do
		end if
		if ctrIteration >= maxIterations then
			return -10002 -- reached max iterations w/o convergence
		end if
		oldM = max(min((oldM+1),newM),(oldM-1))
	until FALSE -- false do-statement
	
	--This is the final result.
	local STRING beforeOutput
	put CRLF & CRLF after beforeOutput
	put "========== FINAL PERSON ESTIMATE ===========" & CRLF after beforeOutput
	put "Rasch Ability Estimate = " & newM & CRLF after beforeOutput
	put "Standard Error         = " & 1/((expectedScoreVariance)^.5) & CRLF after beforeOutput
	put beforeOutput & CRLF & CRLF before text of field "output" of self		

--	local STRING hj_numberFormat
--	hj_numberFormat = "0" & sysDecimal & "00"
--	get newM
--	format number It as hj_numberFormat
--	pMeasure of self = It
--	get (1/((expectedScoreVariance)^.5))
--	format number It as hj_numberFormat
--	pSE of self = It

	pMeasure of self = newM
	pSE of self = (1/((expectedScoreVariance)^.5))

	return -10000 -- dummy return statement, has reached end of program without 
	--expected result
end raschMeasureFromItems

-- konverteringsfunktioner
to get YPUFromMeasure REAL theIn, minMeasure, maxMeasure, maxPU, minPU
	-- theIn sätts till max och min.
	theIn=min(maxMeasure,max(minMeasure,theIn))	
	-- leverer koordinat.
	local LONG output
	output = minPU+((theIn-minMeasure)/(maxMeasure-minMeasure))*(maxPU-minPU)
	return output
end
